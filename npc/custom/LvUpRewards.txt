	function LevelUpReward {
		switch(getarg(0)){
			case 1:
				setd ".bZeny" + getarg(1) + "_" + getarg(2),getarg(3);
				for(.@i=4;.@i<getargcount();.@i += 2){
					.@size = getarraysize(getd(".bItem" + getarg(1) + "_" + getarg(2)));
					setd ".bItem" + getarg(1) + "_" + getarg(2) + "[" + .@size + "]", getarg(.@i);
					setd ".bCount" + getarg(1) + "_" + getarg(2) + "[" + .@size + "]", getarg(.@i + 1);
				}
				break;
			case 2:
				setd ".jZeny" + getarg(1) + "_" + getarg(2),getarg(3);
				for(.@i=4;.@i<getargcount();.@i += 2){
					.@size = getarraysize(getd(".jItem" + getarg(1) + "_" + getarg(2)));
					setd ".jItem" + getarg(1) + "_" + getarg(2) + "[" + .@size + "]", getarg(.@i);
					setd ".jCount" + getarg(1) + "_" + getarg(2) + "[" + .@size + "]", getarg(.@i + 1);
				}
				break;
		}
		return;
	}
end;

OnPCBaseLvUpEvent:
	.@from = (BLvlRewardCheckpoint > 0) ? BLvlRewardCheckpoint + 1 : BaseLevel;
	.@to = BaseLevel;

	for(.@lvl = .@from; .@lvl <= .@to; .@lvl++){
		.@suffix$ = "_" + Class;
		.@suffix0$ = "_0";

		if(getd(".bZeny" + .@lvl + .@suffix$) || getarraysize(getd(".bItem" + .@lvl + .@suffix$)))
			callsub(S_GiveBaseReward, .@lvl, .@suffix$);
		if(getd(".bZeny" + .@lvl + .@suffix0$) || getarraysize(getd(".bItem" + .@lvl + .@suffix0$)))
			callsub(S_GiveBaseReward, .@lvl, .@suffix0$);
	}

	BLvlRewardCheckpoint = .@to;
	end;

S_GiveBaseReward:
	.@lvl = getarg(0);
	.@sfx$ = getarg(1);
	.@zeny = getd(".bZeny" + .@lvl + .@sfx$);
	if(.@zeny) Zeny += .@zeny;

	.@size = getarraysize(getd(".bItem" + .@lvl + .@sfx$));
	for(.@i=0;.@i<.@size;.@i++)
		getitem(getd(".bItem" + .@lvl + .@sfx$ + "[" + .@i + "]"), getd(".bCount" + .@lvl + .@sfx$ + "[" + .@i + "]"));
	return;
