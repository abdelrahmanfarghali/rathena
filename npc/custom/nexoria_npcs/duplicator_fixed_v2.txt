// ---------------------------------------------------------------
// MVP Ladder Tracker: records MVP kills, rewards top 3 at 00:00
// ---------------------------------------------------------------
-	script	MVPLadderTracker	-1,{
OnInit:
	.reward_item = 671;                 // reward item ID
	setarray .reward_amt[0], 3, 2, 1;   // amount for rank 1, 2, 3
	.reset_daily = 1;                   // 1 = wipe ladder after payout
	end;

OnNPCKillEvent:
	if (getmonsterinfo(killedrid, MOB_MVPEXP) <= 0) end;   // MVPs only
	.@cid = getcharid(0);
	.@n = getarraysize($mvp_cid);
	for (.@i = 0; .@i < .@n; .@i++)
		if ($mvp_cid[.@i] == .@cid) break;
	if (.@i == .@n) {                   // first MVP kill: new entry
		$mvp_cid[.@i] = .@cid;
		$mvp_kills[.@i] = 0;
	}
	$mvp_name$[.@i] = strcharinfo(0);   // keeps name current after renames
	$mvp_kills[.@i]++;
	// bubble the entry up to keep the ladder sorted (descending)
	while (.@i > 0 && $mvp_kills[.@i] > $mvp_kills[.@i - 1]) {
		.@j = .@i - 1;
		.@c = $mvp_cid[.@i];     $mvp_cid[.@i] = $mvp_cid[.@j];       $mvp_cid[.@j] = .@c;
		.@k = $mvp_kills[.@i];   $mvp_kills[.@i] = $mvp_kills[.@j];   $mvp_kills[.@j] = .@k;
		.@s$ = $mvp_name$[.@i];  $mvp_name$[.@i] = $mvp_name$[.@j];   $mvp_name$[.@j] = .@s$;
		.@i--;
	}
	end;

OnClock0000:
	.@n = getarraysize($mvp_cid);
	if (.@n <= 0) end;
	for (.@i = 0; .@i < 3 && .@i < .@n; .@i++)
		$mvp_claim[$mvp_cid[.@i]] += .reward_amt[.@i];   // pending, works for offline players
	announce "[MVP Ladder] Daily winner: " + $mvp_name$[0] + " (" + $mvp_kills[0] + " MVPs)! Top 3 can claim rewards at the Statue of the MVP.", bc_all;
	if (.reset_daily) {
		cleararray $mvp_cid[0], 0, .@n;
		cleararray $mvp_kills[0], 0, .@n;
		cleararray $mvp_name$[0], "", .@n;
	}
	end;
}

			$party_board_dungeon$[.@j] = $party_board_dungeon$[.@j+1];
			$party_board_lvl[.@j] = $party_board_lvl[.@j+1];
			$party_board_note$[.@j] = $party_board_note$[.@j+1];
		}
		$party_board_count--;
		mes "[Party Recruiter Board]";
		mes "Your party notice has been removed.";
		close;
	case 4:
		mes "[Party Booking System Guide]";
		mes "Did you know?";
		mes "You can also access the built-in Party Booking window directly from your client:";
		mes " 1. Press ^0055FFAlt + Z^000000 to open the Party Window.";
		mes " 2. Click the ^0055FFBooking^000000 button.";
		mes " 3. Register your party or search for existing groups!";
		close;
	case 5:
		close;
	}
