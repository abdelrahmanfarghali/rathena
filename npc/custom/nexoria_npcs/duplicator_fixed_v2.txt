prontera,170,185,4	script	Mora Coin Exchanger	515,{
	mes "[Mora Coin Exchanger]";
	mes "Welcome, adventurer!";
	mes "I can provide you with ^0055FFMora Coins^000000 for your expeditions into the Laphine territory.";
	next;
	switch(select("Buy Mora Coins (Zeny):Exchange Other Coins:Trade-in Mora Relics (10 Coins):Cancel")) {
	case 1:
		mes "[Mora Coin Exchanger]";
		mes "Mora Coins cost ^0055FF100,000 Zeny^000000 each.";
		mes "How many would you like to purchase?";
		next;
		.@amount = select("1 Mora Coin (100,000z):5 Mora Coins (500,000z):10 Mora Coins (1,000,000z):50 Mora Coins (5,000,000z):Custom Amount:Cancel");
		if (.@amount == 6) close;
		if (.@amount == 1) .@qty = 1;
		else if (.@amount == 2) .@qty = 5;
		else if (.@amount == 3) .@qty = 10;
		else if (.@amount == 4) .@qty = 50;
		else if (.@amount == 5) {
			mes "[Mora Coin Exchanger]";
			mes "Enter the amount of Mora Coins you want (Max 100):";
			input .@qty, 1, 100;
		}
		.@cost = .@qty * 100000;
		if (Zeny < .@cost) {
			mes "[Mora Coin Exchanger]";
			mes "You do not have enough Zeny. You need " + callfunc("F_InsertComma", .@cost) + " Zeny.";
			close;
		}
		if (!checkweight(6380, .@qty)) {
			mes "[Mora Coin Exchanger]";
			mes "You are carrying too much weight.";
			close;
		}
		Zeny -= .@cost;
		getitem 6380, .@qty; // Mora_Coin
		mes "[Mora Coin Exchanger]";
		mes "Here are your " + .@qty + " Mora Coins. May they serve you well!";
		close;
	case 2:
		mes "[Mora Coin Exchanger]";
		mes "I can convert other tokens into Mora Coins:";
		mes " ~ 10 C-Grade Coins -> 1 Mora Coin";
		mes " ~ 1 Silvervine -> 1 Mora Coin";
		next;
		switch(select("Exchange C-Grade Coins (10 -> 1):Exchange Silvervine (1 -> 1):Cancel")) {
		case 1:
			if (countitem(6420) < 10) {
				mes "[Mora Coin Exchanger]";
				mes "You need at least 10 C-Grade Coins.";
				close;
			}
			.@max = countitem(6420) / 10;
			mes "[Mora Coin Exchanger]";
			mes "You have enough C-Grade Coins for up to " + .@max + " Mora Coins.";
			mes "How many would you like?";
			input .@qty, 1, .@max;
			if (countitem(6420) < .@qty * 10) close;
			delitem 6420, .@qty * 10;
			getitem 6380, .@qty;
			mes "[Mora Coin Exchanger]";
			mes "Exchange complete! Obtained " + .@qty + " Mora Coins.";
			close;
		case 2:
			if (countitem(6417) < 1) {
				mes "[Mora Coin Exchanger]";
				mes "You don't have any Silvervine fruit.";
				close;
			}
			.@max = countitem(6417);
			mes "[Mora Coin Exchanger]";
			mes "How many Silvervine would you like to exchange?";
			input .@qty, 1, .@max;
			if (countitem(6417) < .@qty) close;
			delitem 6417, .@qty;
			getitem 6380, .@qty;
			mes "[Mora Coin Exchanger]";
			mes "Exchange complete! Obtained " + .@qty + " Mora Coins.";
			close;
		case 3:
			close;
		}
	case 3:
		mes "[Mora Coin Exchanger]";
		mes "You can return Mora equipment here for ^0055FF10 Mora Coins^000000 each.";
		next;
		setarray .@mora_eq[0],
			2475,2476,2574,2575, // Ur / Peuz
			2477,2478,2577,2578, // Sapha / Nab
			2479,2480,2580,2581, // White / Black Wing
			2481,2482,2582,2583, // Golden Rod / Aqua
			2881,2882,2883,2884, // Mora Rings
			2885,2886,2887,2888;
		.@found = 0;
		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.@mora_eq); .@i++) {
			if (countitem(.@mora_eq[.@i]) > 0) {
				.@menu$ = .@menu$ + getitemname(.@mora_eq[.@i]) + ":";
				.@found_items[.@found] = .@mora_eq[.@i];
				.@found++;
			}
		}
		if (.@found == 0) {
			mes "[Mora Coin Exchanger]";
			mes "You don't have any eligible Mora equipment in your inventory.";
			close;
		}
		.@sel = select(.@menu$ + "Cancel") - 1;
		if (.@sel >= .@found) close;
		.@item_to_trade = .@found_items[.@sel];
		mes "[Mora Coin Exchanger]";
		mes "Trade in 1x " + getitemname(.@item_to_trade) + " for 10 Mora Coins?";
		if (select("Yes:No") == 2) close;
		if (countitem(.@item_to_trade) < 1) close;
		delitem .@item_to_trade, 1;
		getitem 6380, 10;
		mes "[Mora Coin Exchanger]";
		mes "Thank you! Here are your 10 Mora Coins.";
		close;
	case 4:
		close;
	}
}

	mes "[Instance Cooldowns]";
	mes "Greetings, adventurer!";
	mes "I track the temporal stability and cooldowns for all major memorial dungeons.";
	next;
	mes "[Memorial Dungeon Status]";
	function check_cooldown {
		.@qid = getarg(0);
		.@name$ = getarg(1);
		.@status = checkquest(.@qid, PLAYTIME);
		if (.@status == -1 || .@status == 2) {
			mes " ~ " + .@name$ + ": ^00AA00[READY]^000000";
		} else {
			mes " ~ " + .@name$ + ": ^FF0000[COOLDOWN]^000000";
		}
		return;
	}

	check_cooldown(60201, "Endless Tower");
	check_cooldown(12317, "Old Glast Heim");
	check_cooldown(12325, "Faceworm's Nest");
	check_cooldown(7605, "Temple of Demon God");
	check_cooldown(12330, "Horror Toy Factory");
	if (GeffenTournament > gettimetick(2)) {
		.@diff = GeffenTournament - gettimetick(2);
		.@hrs = .@diff / 3600;
		.@mins = (.@diff % 3600) / 60;
		mes " ~ Geffen Tournament: ^FF0000[COOLDOWN (" + .@hrs + "h " + .@mins + "m)]^000000";
	} else {
		mes " ~ Geffen Tournament: ^00AA00[READY]^000000";
	}
	check_cooldown(13185, "Charleston Crisis");
	check_cooldown(12347, "Central Laboratory");
	check_cooldown(12310, "Sara's Memory");
	check_cooldown(12313, "Ghost Palace");
	next;
	if (getgmlevel() >= 99) {
		mes "[GM Cooldown Reset]";
		mes "You possess GM privileges. Would you like to reset your instance cooldowns for testing?";
		if (select("Keep Cooldowns:Reset All My Cooldowns") == 2) {
			erasequest 60200; erasequest 60201;
			erasequest 12317; erasequest 12322;
			erasequest 12325;
			erasequest 7605;
			erasequest 12330; erasequest 12331;
			GeffenTournament = 0;
			erasequest 13185;
			erasequest 12347;
			erasequest 12310;
			erasequest 12313;
			mes "[Instance Cooldowns]";
			mes "All instance cooldowns have been reset.";
		}
	}
	close;
}

		mes "[Post Party Notice]";
		mes "Enter minimum Base Level required (1 - 255):";
		input .@req_lvl, 1, 255;
		mes "[Post Party Notice]";
		mes "Enter a brief note (roles needed, e.g. 'Need AB + Sura'):";
		input .@note$;
		if (.@idx == -1) {
			.@idx = $party_board_count;
			$party_board_count++;
		}
		$party_board_pid[.@idx] = .@pid;
		$party_board_pname$[.@idx] = .@pname$;
		$party_board_leader$[.@idx] = strcharinfo(0);
		$party_board_dungeon$[.@idx] = .@target$;
		$party_board_lvl[.@idx] = .@req_lvl;
		$party_board_note$[.@idx] = .@note$;
		mes "[Party Recruiter Board]";
		mes "Your party notice for ^0055FF" + .@target$ + "^000000 has been successfully posted!";
		close;
	case 3:
		.@pid = getcharid(1);
		if (.@pid < 1) {
			mes "[Party Recruiter Board]";
			mes "You are not in a party.";
			close;
		}
		.@found = -1;
		for (.@i = 0; .@i < $party_board_count; .@i++) {
			if ($party_board_pid[.@i] == .@pid) {
				.@found = .@i;
				break;
			}
		}
		if (.@found == -1) {
			mes "[Party Recruiter Board]";
			mes "Your party does not currently have an active notice posted.";
			close;
		}
		if (is_party_leader() == false) {
			mes "[Party Recruiter Board]";
			mes "Only the Party Leader can remove the recruitment notice.";
			close;
		}
		for (.@j = .@found; .@j < $party_board_count - 1; .@j++) {
			$party_board_pid[.@j] = $party_board_pid[.@j+1];
			$party_board_pname$[.@j] = $party_board_pname$[.@j+1];
			$party_board_leader$[.@j] = $party_board_leader$[.@j+1];
			$party_board_dungeon$[.@j] = $party_board_dungeon$[.@j+1];
			$party_board_lvl[.@j] = $party_board_lvl[.@j+1];
			$party_board_note$[.@j] = $party_board_note$[.@j+1];
		}
		$party_board_count--;
		mes "[Party Recruiter Board]";
		mes "Your party notice has been removed.";
		close;
	case 4:
		mes "[Party Booking System Guide]";
		mes "Did you know?";
		mes "You can also access the built-in Party Booking window directly from your client:";
		mes " 1. Press ^0055FFAlt + Z^000000 to open the Party Window.";
		mes " 2. Click the ^0055FFBooking^000000 button.";
		mes " 3. Register your party or search for existing groups!";
		close;
	case 5:
		close;
	}
}
