// ============================================================
// Refine Ticket Exchanger
// ============================================================
prontera,142,175,4	script	Refine Ticket Exchanger	726,{
	// ---------- Config ----------
	.@gem     = 671;   // Gemstone item ID
	.@cap_hrs = 48;    // max AFK hours that can accumulate
	.@min_lv  = 6;     // lowest ticket level
	.@max_lv  = 19;    // highest ticket level
	.@base    = 15;    // cost at .@min_lv
	.@step    = 5;     // added cost per level

	// Ticket IDs indexed by refine level (+6..+19)
	setarray .@tkw[6], 6231,6230,6229,6228,6993,6238,6584,6870,6871,6872,6873,6874,6875,6864;
	setarray .@tka[6], 6235,6234,6233,6232,6994,6239,6585,6876,6877,6878,6879,6880,6881,6865;

	// ---------- AFK accrual ----------
	.@now = gettimetick(2);
	if (!TG_LAST) TG_LAST = .@now;
	.@hrs = min((.@now - TG_LAST) / 3600, .@cap_hrs);

	mes "[Refine Ticket Exchanger]";
	mes getitemname(.@gem) + "s: ^0000FF" + countitem(.@gem) + "^000000";
	mes "AFK reward ready: ^00AA00" + .@hrs + "^000000 (+1 per hour)";
	next;
	switch (select("Claim AFK " + getitemname(.@gem) + "s:Exchange " + getitemname(.@gem) + "s for tickets:Refine with tickets:Cancel")) {

	case 1:
		if ((.@hrs//2) < 1) { mes "[Refine Ticket Exchanger]"; mes "Come back later."; close; }
		TG_LAST += .@hrs * 3600;   // keep the remainder
		getitem .@gem, (.@hrs//2);
		mes "[Refine Ticket Exchanger]";
		mes "Received " + .@hrs + " gold coins(s).";
		close;

	case 2:
		.@menu$ = "";
		for (.@i = .@min_lv; .@i <= .@max_lv; .@i++)
			.@menu$ += "+" + .@i + " Ticket (" + (.@base + (.@i - .@min_lv) * .@step) + " gems):";
		.@lv   = .@min_lv - 1 + select(.@menu$);
		.@cost = .@base + (.@lv - .@min_lv) * .@step;
		if (countitem(.@gem) < .@cost) { mes "Not enough " + getitemname(.@gem) + "s."; close; }
		mes "[Refine Ticket Exchanger]";
		mes "Weapon or armor ticket?";
		.@type = select("Weapon:Armor");
		delitem .@gem, .@cost;
		getitem (.@type == 1 ? .@tkw[.@lv] : .@tka[.@lv]), 1;
		close;

	case 3:
		setarray .@parts[0], EQI_HEAD_TOP, EQI_ARMOR, EQI_HAND_R, EQI_HAND_L, EQI_GARMENT, EQI_SHOES;
		.@menu$ = "";
		for (.@i = 0; .@i < getarraysize(.@parts); .@i++)
			.@menu$ += (getequipisequiped(.@parts[.@i]) ? getequipname(.@parts[.@i]) : "-") + ":";
		.@p = .@parts[select(.@menu$) - 1];

		if (!getequipisequiped(.@p) || !getequipisenableref(.@p)) { mes "Cannot refine that."; close; }
		.@cnt = getequiprefinerycnt(.@p);
		.@isw = (.@p == EQI_HAND_R && getequipweaponlv(.@p) > 0);

		// Ticket amount: weapons Lv1-3 = 1, Lv4 = 2; armor normal = 1, higher = 2
		if (.@isw)
			.@amt = (getequipweaponlv(.@p) >= 4) ? 2 : 1;
		else
			.@amt = (getiteminfo(getequipid(.@p), ITEMINFO_ARMORLEVEL) >= 2) ? 2 : 1;

		// Build ticket menu from inventory
		.@menu$ = "";
		.@n = 0;
		for (.@i = .@min_lv; .@i <= .@max_lv; .@i++) {
			if (.@i == .@cnt) continue;
			.@tid = .@isw ? .@tkw[.@i] : .@tka[.@i];
			if (countitem(.@tid) < .@amt) continue;
			.@lvs[.@n] = .@i;
			.@n++;
			.@menu$ += "+" + .@i + " Ticket (" + countitem(.@tid) + " owned)"
			         + (.@i < .@cnt ? " ^FF0000[Downgrade]^000000" : "") + ":";
		}
		if (!.@n) {
			mes "[Refine Ticket Exchanger]";
			mes "You have no usable ticket for this item (" + .@amt + "x needed).";
			close;
		}
		.@to  = .@lvs[select(.@menu$) - 1];
		.@tid = .@isw ? .@tkw[.@to] : .@tka[.@to];

		// Critical confirmation for downgrades
		if (.@to < .@cnt) {
			mes "[Refine Ticket Exchanger]";
			mes "^FF0000WARNING:^000000 this will set your";
			mes getequipname(.@p);
			mes "from ^FF0000+" + .@cnt + "^000000 DOWN to ^FF0000+" + .@to + "^000000.";
			mes "This cannot be undone.";
			next;
			if (select("Yes, downgrade it:Cancel") != 1) close;
			// Re-validate after the prompt
			if (!getequipisequiped(.@p) || getequiprefinerycnt(.@p) != .@cnt) { mes "Item changed. Aborted."; close; }
		}
		if (countitem(.@tid) < .@amt) { mes "Ticket missing."; close; }

		delitem .@tid, .@amt;
		while (getequiprefinerycnt(.@p) < .@to) successrefitem .@p;
		while (getequiprefinerycnt(.@p) > .@to) downrefitem .@p;
		mes "[Refine Ticket Exchanger]";
		mes "Done. Your item is now ^0000FF+" + getequiprefinerycnt(.@p) + "^000000.";
		close;

	case 4:
		close;
	}
}
