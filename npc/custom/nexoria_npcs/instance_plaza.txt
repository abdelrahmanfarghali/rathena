// ====================================================================
// Instance Generation & Entry Engine
// Two-phase flow (mirrors official rAthena):
//   Phase 1 (leader only): instance_create  -> reserves the instance,
//                           triggers OnInstanceInit inside the instance script
//                           which spawns all NPCs and mobs.
//   Phase 2 (all members): instance_enter   -> warps player into the
//                           already-initialised instance map.
// ====================================================================
GenerateInstance:
    // --- PHASE 2: Party member entry (instance already exists) ---
    // Non-leaders, or anyone touching the portal after the leader created it,
    // skip straight to entry.
    if (instance_id(1) > 0) {
        mes "[ ^FF6600Instance Manager^0000 ]";
        mes "An instance is ready. Entering ^0055FF" + instance_live_info(ILI_NAME, instance_id(1)) + "^0000...";
        close2;
        callsub L_Enter;
        end;
    }

    // --- PHASE 1: Only the leader can create a fresh instance ---
    if (!is_party_leader()) {
        mes "[ ^FF6600Instance Manager^0000 ]";
        mes "Only the ^FF0000Party Leader^0000 can generate a new instance.";
        mes "Ask your leader to touch the portal first to create it.";
        mes "Once created, touch it again to enter.";
        close;
    }

    mes "[ ^FF6600Instance Manager^0000 ]";
    mes "Generating ^0055FF" + .@inst_name$ + "^0000...";
    mes "After creation, touch the portal again to enter.";

    if (instance_create(.@inst_name$) < 0) {
        mes " ";
        mes "^FF0000Failed to create instance.^0000";
        mes "The server might be full, or your party is still on cooldown.";
        close;
    }

    // Instance created successfully. OnInstanceInit inside the instance
    // script will now run automatically, spawning all NPCs and mobs.
    mes " ";
    mes "^00AA00Instance reserved!^0000 All party members may now touch the portal to enter.";
    close;

// ====================================================================
// Entry subroutine — handles all IE_* error codes from instance_enter
// ====================================================================
L_Enter:
    switch (instance_enter(.@inst_name$)) {
    case IE_OK:
        end;
    case IE_NOINSTANCE:
        mes "[ ^FF6600Instance Manager^0000 ]";
        mes "No instance exists yet.";
        mes "The ^FF0000Party Leader^0000 must touch the portal first to generate one.";
        close;
    case IE_NOMEMBER:
        mes "[ ^FF6600Instance Manager^0000 ]";
        mes "You must be in a party to enter.";
        close;
    case IE_OTHER:
    default:
        mes "[ ^FF6600Instance Manager^0000 ]";
        mes "^FF0000An unknown error occurred while entering.^0000";
        mes "Please try again or contact a GM.";
        close;
    }
    return;
