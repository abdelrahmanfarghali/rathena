// ====================================================================
// Instance Plaza Portal (data-driven registry)
// S_Reg args: "<instance name>", <category>, <gateX>, <gateY>
//   gateX/gateY = 0,0 -> generic instance (create/enter via this portal)
//   gateX/gateY > 0   -> NPC-managed instance (warp to its gate NPC)
// ====================================================================
prontera,165,167,0	script	Instance Plaza Portal	PORTAL,{

OnTouch:
    if (!getcharid(1)) {
        message strcharinfo(0), "You must form a party to enter the Instance Plaza.";
        end;
    }

    // Party already has an active instance: enter or destroy
    if (instance_id(IM_PARTY) > 0) {
        .@active$ = instance_live_info(ILI_NAME, instance_id(IM_PARTY));
        mes "[ ^FF6600Instance Manager^000000]";
        mes "Your party already has an active instance running:";
        mes "^0055FF" + (.@active$ != "" ? .@active$ : "Active Instance") + "^000000";
        next;
        switch (select("Enter Instance:Destroy Instance:Cancel")) {
        case 1:
            callsub L_Enter;
            end;
        case 2:
            if (!is_party_leader()) {
                mes "[ ^FF6600Instance Manager^000000]";
                mes "Only the ^FF0000Party Leader^000000can destroy the active instance.";
                close;
            }
            instance_destroy instance_id(IM_PARTY);
            mes "[ ^FF6600Instance Manager^000000]";
            mes "Instance destroyed. You may now generate a new one.";
            close;
        case 3:
            close;
        }
    }

    // Category menu
    mes "[ ^FF6600Instance Manager^000000]";
    mes "Select an instance category to browse:";
    next;
    .@menu$ = "";
    for (.@i = 0; .@i < getarraysize(.cat_name$); .@i++)
        .@menu$ += .cat_name$[.@i] + ":";
    .@c = select(.@menu$ + "Cancel");
    if (.@c > getarraysize(.cat_name$)) close;

    // Instance menu for the chosen category
    .@menu$ = "";
    .@n = 0;
    for (.@i = 0; .@i < getarraysize(.inst_name$); .@i++) {
        if (.inst_cat[.@i] != .@c) continue;
        .@idx[.@n] = .@i;
        .@menu$ += .inst_name$[.@i] + ":";
        .@n++;
    }
    mes "[ ^FF6600Instance Manager^000000]";
    mes "Select: ^0055FF" + .cat_name$[.@c - 1] + "^000000";
    next;
    .@s = select(.@menu$ + "Back to Categories") - 1;
    if (.@s >= .@n) goto OnTouch;
    .@inst = .@idx[.@s];
    .@inst_name$ = .inst_name$[.@inst];

    // ---------------- Entrance choice ----------------
    // NPC-managed: warp to its gate NPC
    if (.inst_gx[.@inst] > 0) {
        mes "[ ^FF6600Instance Manager^000000]";
        mes "^0055FF" + .@inst_name$ + "^000000";
        mes "This instance is started by its own gate NPC.";
        next;
        switch (select("Go to the gate:Back to categories:Cancel")) {
        case 1:
            close2;
            warp "prontera", .inst_gx[.@inst], .inst_gy[.@inst];
            end;
        case 2:
            goto OnTouch;
        case 3:
            close;
        }
    }

    // Generic: create (leader) / enter (members)
    mes "[ ^FF6600Instance Manager^000000]";
    mes "^0055FF" + .@inst_name$ + "^000000";
    next;
    switch (select("Create instance (Party Leader):Enter instance:Back to categories:Cancel")) {
    case 1:
        if (!is_party_leader()) {
            mes "[ ^FF6600Instance Manager^000000]";
            mes "Only the ^FF0000Party Leader^000000can generate a new instance.";
            close;
        }
        if (instance_create(.@inst_name$, getcharid(1), IM_PARTY) < 0) {
            mes "[ ^FF6600Instance Manager^000000]";
            mes "^FF0000Failed to create instance.^000000";
            mes "The server might be full, or your party is still on cooldown.";
            close;
        }
        mes "[ ^FF6600Instance Manager^000000]";
        mes "^00AA00Instance reserved!^000000All party members may now touch the portal to enter.";
        close;
    case 2:
        callsub L_Enter;
        end;
    case 3:
        goto OnTouch;
    case 4:
        close;
    }
    end;

// --------------------------------------------------------------------
// Entry subroutine (instance name read from the party's live instance)
// --------------------------------------------------------------------
L_Enter:
    if (instance_id(IM_PARTY) <= 0) {
        mes "[ ^FF6600Instance Manager^000000]";
        mes "No instance exists yet.";
        mes "The ^FF0000Party Leader^000000must create one first.";
        close;
    }
    switch (instance_enter(instance_live_info(ILI_NAME, instance_id(IM_PARTY)))) {
    case IE_OK:
        close;
    case IE_NOINSTANCE:
        mes "[ ^FF6600Instance Manager^000000]";
        mes "No instance exists yet.";
        close;
    case IE_NOMEMBER:
        mes "[ ^FF6600Instance Manager^000000]";
        mes "You must be in a party to enter.";
        close;
    default:
        mes "[ ^FF6600Instance Manager^000000]";
        mes "^FF0000An unknown error occurred while entering.^000000";
        mes "Please try again or contact a GM.";
        close;
    }
    end;

// --------------------------------------------------------------------
// Registry helper
// --------------------------------------------------------------------
S_Reg:
    .@k = getarraysize(.inst_name$);
    .inst_name$[.@k] = getarg(0);
    .inst_cat[.@k]   = getarg(1);
    .inst_gx[.@k]    = getarg(2);
    .inst_gy[.@k]    = getarg(3);
    return;

OnInit:
    setarray .cat_name$[0],
        "Classic & Iconic Dungeons",
        "Heroes' Trails & Dimensional Gap",
        "Episode 16";
	//, "Episode 17 (Banquet / OS)",
        //"Episode 18 & High-End Raids",
        //"Special & Event Dungeons";

    // Category 1
    callsub S_Reg, "Endless Tower", 1, 0, 0;
    callsub S_Reg, "Old Glast Heim", 1, 0, 0;
    callsub S_Reg, "Old Glast Heim (Beginner)", 1, 0, 0;
    callsub S_Reg, "Old Glast Heim (Challenge Mode)", 1, 0, 0;
    callsub S_Reg, "Fall of Glast Heim (Normal)", 1, 0, 0;
    callsub S_Reg, "Fall of Glast Heim (Advanced)", 1, 0, 0;
    callsub S_Reg, "Sealed Catacomb", 1, 0, 0;
    callsub S_Reg, "Orc's Memory", 1, 0, 0;
    callsub S_Reg, "Nidhoggur's Nest", 1, 0, 0;
    callsub S_Reg, "Culvert", 1, 0, 0;
    callsub S_Reg, "Octopus Cave", 1, 0, 0;
    callsub S_Reg, "Mistwood Maze", 1, 0, 0;
    callsub S_Reg, "Eclage Interior", 1, 0, 0;
    callsub S_Reg, "Wolfchev's Laboratory", 1, 0, 0;
    callsub S_Reg, "Bakonawa Lake", 1, 0, 0;
    callsub S_Reg, "Bangungot Hospital 2F", 1, 0, 0;
    callsub S_Reg, "Buwaya Cave", 1, 0, 0;

    // Category 2 (gate coords = 2 cells west of each duplicate NPC)
    callsub S_Reg, "Sara's Memories", 2, 0, 0;
    callsub S_Reg, "Geffen Magic Tournament", 2, 163, 144;
    callsub S_Reg, "Ghost Palace", 2, 0, 0;
    callsub S_Reg, "Devil's Tower", 2, 0, 0;
    callsub S_Reg, "Faceworm's Nest", 2, 0, 0;
    callsub S_Reg, "Horror Toy Factory", 2, 163, 141;
    callsub S_Reg, "Fenrir and Sarah", 2, 0, 0;
    callsub S_Reg, "Assault on the Airship", 2, 0, 0;
    callsub S_Reg, "Nightmarish Jitterbug", 2, 0, 0;
    callsub S_Reg, "Isle of Bios", 2, 0, 0;
    callsub S_Reg, "Morse's Cave", 2, 0, 0;
    callsub S_Reg, "Temple of the Demon God", 2, 0, 0;
    callsub S_Reg, "Central Laboratory", 2, 163, 138;
    callsub S_Reg, "Last room", 2, 0, 0;
    callsub S_Reg, "Charleston in Distress", 2, 0, 0;
    callsub S_Reg, "Infinite Space", 2, 0, 0;

    // Category 3
    callsub S_Reg, "Ritual of Blessing", 3, 0, 0;
    callsub S_Reg, "Room of Consciousness", 3, 0, 0;
    callsub S_Reg, "Sky Fortress Invasion", 3, 0, 0;
    //callsub S_Reg, "Heart Hunter War Base 1", 4, 0, 0;
    //callsub S_Reg, "Heart Hunter War Base 2", 4, 0, 0;
    //callsub S_Reg, "Werner Laboratory central room#1", 4, 0, 0;
    //callsub S_Reg, "Werner Laboratory central room#2", 4, 0, 0;
    //callsub S_Reg, "Regenschirm", 4, 0, 0;
    //callsub S_Reg, "Sealed OS", 4, 0, 0;
    //callsub S_Reg, "OS Occupation", 4, 0, 0;
    //callsub S_Reg, "2nd OS Search", 4, 0, 0;
    //callsub S_Reg, "Cor Memorial", 4, 0, 0;
    //callsub S_Reg, "Half Moon In The Daylight", 4, 0, 0;

    // Category 4
    //callsub S_Reg, "Thanatos Tower", 4, 0, 0;
    //callsub S_Reg, "Constellation Tower", 4, 0, 0;
    //callsub S_Reg, "Sunken Tower", 4, 0, 0;
    //callsub S_Reg, "Tomb of Remorse", 4, 0, 0;

    //callsub S_Reg, "Airship Destruction", 4, 0, 0;

    //callsub S_Reg, "Thor Military Base", 4, 0, 0;
    //callsub S_Reg, "Wolves Gathering Place", 4, 0, 0;
    //callsub S_Reg, "Sanctuary Purification", 4, 0, 0;
    //callsub S_Reg, "High Priest's Villa", 4, 0, 0;
    //callsub S_Reg, "Villa of Deception", 4, 0, 0;

    //callsub S_Reg, "Confused Snake's Nest", 4, 0, 0;
