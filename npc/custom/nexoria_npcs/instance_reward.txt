// ---------------- Reward Function ----------------
// Usage: callfunc "F_InstanceCoinReward", <amount>;
// Call it from the final boss's OnMyMobDead label (killer must be attached).
//OnMyMobDead:
//	callfunc "F_InstanceCoinReward", 5;
//	end;

function	script	F_InstanceCoinReward	{
	.@amount = getarg(0, 5);
	.@self   = getcharid(3);
	.@map$   = strcharinfo(PC_MAP);
	.@party  = getcharid(1);

	if (.@party) {
		getpartymember .@party, 2;                 // account IDs of online members
		.@count = $@partymembercount;
		copyarray .@aid[0], $@partymemberaid[0], .@count;
	} else {
		.@aid[0] = .@self;                         // solo player
		.@count = 1;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@aid[.@i]))
			continue;
		if (strcharinfo(PC_MAP) != .@map$)         // only players still in the instance
			continue;
		if (checkweight(674, .@amount) == 0) {
			dispbottom "Instance reward failed: you are carrying too much.";
			continue;
		}
		getitem 674, .@amount;
		dispbottom "You received " + .@amount + " Instance Coin(s).";
	}

	attachrid(.@self);                             // restore original player context
	return;
}
