// ---------------- Reward Function ---------------- //
// Usage: callfunc "F_InstanceCoinReward", <amount>;
// Call it from the final boss's OnMyMobDead label (killer must be attached).
//
// OnMyMobDead:
//     callfunc "F_InstanceCoinReward", 5;
//     end;
function	script	F_InstanceCoinReward	{
	.@amount = getarg(0, 5);
	.@item = 30000;
	.@flag$ = "'reward_" + getcharid(0); // instance-scoped, per character, resets every new run

	if (getd(.@flag$)) {
		mes "You have already claimed your reward.";
		return 0;
	}
	if (checkweight(.@item, .@amount) == 0) {
		mes "You are carrying too much. Free some space and talk to me again.";
		return 0; // flag untouched: can retry
	}

	setd .@flag$, 1; // mark BEFORE giving, so it can't be claimed twice
	getitem .@item, .@amount;
	mes "You received " + .@amount + "x " + getitemname(.@item) + ".";
	return 1;
}
